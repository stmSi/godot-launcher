cmake_minimum_required(VERSION 3.25)
project(GodotLauncher VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(GL_DEFAULT_FRAMEWORK "${CMAKE_CURRENT_SOURCE_DIR}/deps/af-framework")
set(GL_DEFAULT_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/deps/third_party")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../Arcflow/framework/CMakeLists.txt")
    set(GL_DEFAULT_FRAMEWORK "${CMAKE_CURRENT_SOURCE_DIR}/../Arcflow/framework")
    set(GL_DEFAULT_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../Arcflow/third_party")
endif()
set(ARCFLOW_FRAMEWORK_SOURCE_DIR "${GL_DEFAULT_FRAMEWORK}" CACHE PATH "AF Framework checkout")
set(AF_FRAMEWORK_DEPENDENCY_ROOT "${GL_DEFAULT_DEPS}" CACHE PATH "Framework dependencies")
set(AF_FRAMEWORK_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(AF_FRAMEWORK_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${ARCFLOW_FRAMEWORK_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Set ARCFLOW_FRAMEWORK_SOURCE_DIR to your af-framework checkout")
endif()
if(APPLE)
    set(AF_FRAMEWORK_BUILD_VULKAN OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory("${ARCFLOW_FRAMEWORK_SOURCE_DIR}" framework EXCLUDE_FROM_ALL)
find_package(CURL 7.85 REQUIRED)
find_package(LibArchive REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(json-c CONFIG QUIET)
if(TARGET json-c::json-c)
    set(GL_JSON_TARGET json-c::json-c)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
    set(GL_JSON_TARGET PkgConfig::JSONC)
endif()
add_library(launcher_core STATIC src/core.c)
target_include_directories(launcher_core PUBLIC src)
target_link_libraries(launcher_core PUBLIC CURL::libcurl LibArchive::LibArchive OpenSSL::Crypto ${GL_JSON_TARGET})
if(NOT WIN32)
    target_compile_definitions(launcher_core PRIVATE _POSIX_C_SOURCE=200809L)
endif()
add_executable(godot-launcher src/main.c src/app.c src/render.c)
target_link_libraries(godot-launcher PRIVATE launcher_core af::framework)
if(WIN32)
    enable_language(RC)
    target_sources(godot-launcher PRIVATE packaging/launcher.rc)
    target_include_directories(godot-launcher PRIVATE packaging)
    set_target_properties(godot-launcher PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
if(APPLE)
    target_link_libraries(godot-launcher PRIVATE "-framework CoreGraphics")
    set_target_properties(godot-launcher PROPERTIES
        MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_GUI_IDENTIFIER io.arcflow.GodotLauncher
        MACOSX_BUNDLE_BUNDLE_NAME "Godot Launcher"
        MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
        MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}")
    set(GL_RESOURCE_DIR "$<TARGET_BUNDLE_DIR:godot-launcher>/Contents/Resources")
else()
    set(GL_RESOURCE_DIR "$<TARGET_FILE_DIR:godot-launcher>")
endif()
target_compile_definitions(godot-launcher PRIVATE GL_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
foreach(target launcher_core godot-launcher)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
add_custom_command(TARGET godot-launcher POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "${GL_RESOURCE_DIR}/assets"
)
if(NOT APPLE)
    add_custom_command(TARGET godot-launcher POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${AF_FRAMEWORK_SHADER_OUTPUT_DIR}" "$<TARGET_FILE_DIR:godot-launcher>/shaders")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(launcher-tests tests/test_core.c)
    target_link_libraries(launcher-tests PRIVATE launcher_core)
    add_test(NAME launcher-core COMMAND launcher-tests)
endif()
install(TARGETS godot-launcher RUNTIME DESTINATION bin BUNDLE DESTINATION .)
if(APPLE)
    configure_file(packaging/macos-install.cmake.in macos-install.cmake @ONLY)
    install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/macos-install.cmake")
endif()
if(NOT APPLE)
    install(DIRECTORY assets/ DESTINATION bin/assets)
endif()
if(NOT APPLE)
    install(DIRECTORY "${AF_FRAMEWORK_SHADER_OUTPUT_DIR}/" DESTINATION bin/shaders)
endif()
if(UNIX AND NOT APPLE)
    install(FILES packaging/io.arcflow.GodotLauncher.desktop DESTINATION share/applications)
    install(FILES assets/launcher.svg DESTINATION share/icons/hicolor/scalable/apps RENAME io.arcflow.GodotLauncher.svg)
endif()
